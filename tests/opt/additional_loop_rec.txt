fn looper := (i32 n) -> i32
{
    i32 mut i{0}
    while i != n
    {
        i := i + 1
    }
    exit i
}
fn rec := (i32 n) -> i32
{
    if n == 0
    {
        exit 0
    }
    i32 prev{n - 1}
    exit rec(prev)
}
fn pure := (i32 n) -> i32
{
    i32 res{n + 1}
    exit res
}
i32 a{looper(3)}
i32 b{rec(3)}
i32 c{pure(3)}
